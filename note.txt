query = '''
any where host.os.type == "windows" and  
(
  process.name: "reg.exe" and  process.parent.name : (
    "powershell.exe",
    "pwsh.exe",
    "cmd.exe",
    "wscript.exe",
    "cscript.exe",
    "mshta.exe",
    "rundll32.exe",
    "regsvr32.exe",
    "wmic.exe"
  ) or 
  process.command_line: ("*New-ItemProperty*", "*New-Item*", "*Set-ItemProperty*")
) and 
process.command_line: "*SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\App Paths*.exe*"

FROM logs-windows.sysmon_operational-*
| WHERE host.os.type == "windows"
    AND event.code == "1"
    AND process.name == "reg.exe"
    AND TO_LOWER(process.command_line) LIKE "*app paths*"
    AND TO_LOWER(process.command_line) LIKE "*/d*"
| GROK process.command_line """(?i).*App Paths\\+%{DATA:app_name}" /t REG_SZ /d %{NOTSPACE:target_path} /f.*"""
| EVAL app_name_norm = TO_LOWER(app_name),
       target_name = TO_LOWER(REPLACE(target_path, """^.*\\""", ""))
| WHERE app_name_norm != target_name
| KEEP @timestamp, host.name, user.name,
       process.parent.name, process.command_line,
       app_name, target_path, target_name
